# ⚠ UN GROUPE N'HERITE PAS DES REGLES DU GROUPE `*`. C'est la regle la plus mal
#   comprise de ce fichier, et elle a deja mordu deux fois ici : un agent qui
#   trouve son propre `User-agent:` ignore ENTIEREMENT le bloc `*`. Tout groupe
#   nomme doit donc repeter la liste des chemins prives, sans exception.
#
# 🔴 CE QUE ÇA CORRIGE CETTE FOIS. Les quatre groupes d'IA ci-dessous n'avaient
#    QUE `Allow: /`, donc aucun Disallow : ils etaient les seuls autorises a
#    explorer /compte, /auth et /bienvenue — exactement l'erreur que ce fichier
#    documentait deja pour SemrushBot, repetee sur quatre autres groupes.
#    Et `/user/` n'etait interdit NULLE PART.

User-agent: *
Allow: /
Disallow: /compte
Disallow: /auth
Disallow: /bienvenue
Disallow: /parametres
Disallow: /messages
Disallow: /notifications
# 🔴 LES PAGES DE MEMBRE NE S'INDEXENT PAS. Un profil porte ce qu'une personne
#    a publie, ou elle est allee et ce qu'elle a goute : c'est une trace de
#    deplacements. Elle doit rester trouvable SUR le site, pas moissonnable
#    depuis l'exterieur.
#    ⚠ On protege par defaut parce que le temps n'est pas symetrique : on peut
#      ouvrir plus tard, on ne peut pas desindexer ce qui a deja ete aspire.
#      Les etablissements restent ouverts — c'est la surface commerciale, et
#      c'est /p/ qui la porte.
Disallow: /user/
# La recherche produit une infinite d'URL par combinaison de filtres : la
# laisser explorer dilue le domaine en pages quasi identiques.
Disallow: /recherche?
